NAME: Slack-RegistrationModel

DEFINITION
----------
  type: object
  description: One configured Workspace notification destination.
  properties:
    id: optional
      type: integer
      example: 1
    provider: optional
      type: string
      example: slack
    eventType: optional
      type: string
      description: One of the values returned by WorkspaceNotificationConfigModel.eventTypes.
      example: BIRTHDAY
    webhookUrl: optional
      type: ["string","null"]
      description: Masked URL — only the workspace + channel-id segments are exposed; the secret is replaced with `…`. Null when no webhook is stored.
      example: https://hooks.slack.com/services/TH6MG0V7D/B0AU672L2LA/…
    channelLabel: optional
      type: ["string","null"]
      example: #hr-team
    subunits: optional
      type: array
      description: Multi-subunit filter. Empty array means 'all employees, no filter'.
      items:
        type: object
        properties:
          id: optional
            type: integer
            example: 4
          name: optional
            type: string
            example: Engineering
    timezone: optional
      type: string
      example: Asia/Colombo
    dailySendTime: optional
      type: string
      example: 09:00
    active: optional
      type: boolean
      example: true
